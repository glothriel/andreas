{{- /* A taped photo with a slow slideshow of ~5 images spread across the gallery. */ -}} {{- $g := .g -}} {{- $imgs := $g.Resources.ByType "image" -}} {{- $n := len $imgs -}} {{- $step := math.Max 1 (math.Ceil (div (float $n) 5)) | int -}} {{- $picks := slice (partial "cover.html" $g) -}} {{- range $j, $img := $imgs }}{{ if and (eq (mod (add $j 1) $step) 0) (lt (len $picks) 5) (ne $img (index $picks 0)) }}{{ $picks = $picks | append $img }}{{ end }}{{ end -}} {{- range $k, $p := $picks }} {{- $t := $p.Resize "900x webp q72" }} {{- end }} {{ $g.Title }}{{ $n }}
